#!/bin/bash

echo "====================================="
echo " Big Data Pipeline - Server Launcher"
echo "====================================="

echo ""
echo "[1/3] Starting PostgreSQL..."
sudo systemctl start postgresql
echo "PostgreSQL started."

echo ""
echo "[2/3] Starting HDFS..."
start-dfs.sh
echo "HDFS started."

echo ""
echo "[3/3] Starting YARN..."
start-yarn.sh
echo "YARN started."

echo ""
echo "Verifying running processes..."
jps

echo ""
echo "====================================="
echo " Choose which Spark server to start:"
echo " 1) Spark Connect (for Jupyter notebook - port 15005)"
echo " 2) Thrift Server (for Power BI - port 10000)"
echo "====================================="
read -p "Enter 1 or 2: " choice

if [ "$choice" == "1" ]; then
    echo ""
    echo "Starting Spark Connect on port 15005..."
    $SPARK_HOME/sbin/start-connect-server.sh \
      --master yarn \
      --packages org.apache.spark:spark-connect_2.12:3.5.8 \
      --conf spark.connect.grpc.binding.address=0.0.0.0 \
      --conf spark.connect.grpc.binding.port=15005 \
      --conf spark.sql.catalogImplementation=hive \
      --conf spark.sql.warehouse.dir=hdfs://localhost/user/hive/warehouse \
      --conf spark.hadoop.javax.jdo.option.ConnectionURL=jdbc:postgresql://localhost:5432/metastore \
      --conf spark.hadoop.javax.jdo.option.ConnectionDriverName=org.postgresql.Driver \
      --conf spark.hadoop.javax.jdo.option.ConnectionUserName=hiveuser \
      --conf spark.hadoop.javax.jdo.option.ConnectionPassword=hivepassword
    echo ""
    echo "Spark Connect is running."
    echo "Launch Jupyter with: jupyter lab --ip=0.0.0.0 --no-browser --port=8888"

elif [ "$choice" == "2" ]; then
    echo ""
    echo "Stopping Spark Connect if running..."
    $SPARK_HOME/sbin/stop-connect-server.sh
    sleep 10
    echo "Starting Thrift Server on port 10000..."
    $SPARK_HOME/sbin/start-thriftserver.sh \
      --master yarn \
      --deploy-mode client \
      --conf spark.sql.catalogImplementation=hive \
      --conf spark.sql.warehouse.dir=hdfs://localhost/user/hive/warehouse \
      --conf spark.hadoop.javax.jdo.option.ConnectionURL=jdbc:postgresql://localhost:5432/metastore \
      --conf spark.hadoop.javax.jdo.option.ConnectionDriverName=org.postgresql.Driver \
      --conf spark.hadoop.javax.jdo.option.ConnectionUserName=hiveuser \
      --conf spark.hadoop.javax.jdo.option.ConnectionPassword=hivepassword \
      --hiveconf hive.server2.thrift.bind.host=0.0.0.0 \
      --hiveconf hive.server2.thrift.port=10000
    echo ""
    echo "Thrift Server is running."
    echo "Connect Power BI to: 192.168.1.16:10000 (Standard protocol)"
else
    echo "Invalid choice. Exiting."
    exit 1
fi

echo ""
echo "Done. Use 'jps' to verify all processes."
